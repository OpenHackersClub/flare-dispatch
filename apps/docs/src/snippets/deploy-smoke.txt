import { Effect, Schema } from "effect";
import { defineRun, step, sandbox } from "@flaredispatch/core";

// Webhook mode: fires on deployment_status.success and probes the
// live URLs in parallel — zero GHA minutes, no workflow file.
export const deploySmoke = defineRun({
  name: "deploy-smoke",
  version: "1.0.0",

  triggers: [
    {
      event: "deployment_status",
      gate: ({ payload }) => payload.deployment_status.state === "success",
      idempotencyKey: ({ payload }) =>
        `deploy-smoke:${payload.deployment.id}`,
      inputs: ({ payload }) => ({
        baseURL: payload.deployment_status.environment_url,
        paths: ["/", "/health", "/api/status"],
      }),
    },
  ],

  inputs: Schema.Struct({
    baseURL: Schema.String,
    paths: Schema.Array(Schema.String),
  }),
  outputs: Schema.Struct({ failed: Schema.Number }),
  limits: { maxDurationSec: 300 },

  run: (input) =>
    Effect.gen(function* () {
      // Probe every critical URL in parallel once the deploy lands.
      const results = yield* step("probe", () =>
        Effect.forEach(
          input.paths,
          (path) =>
            sandbox.exec({
              command: ["curl", "-sS", "-o", "/dev/null",
                        "-w", "%{http_code}", `${input.baseURL}${path}`],
            }),
          { concurrency: input.paths.length },
        ),
      );
      const failed = results.filter(
        (r) => !(Number(r.stdout) >= 200 && Number(r.stdout) < 400),
      ).length;
      return { failed };
    }),
});
