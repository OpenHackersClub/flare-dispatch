import { Effect, Schema } from "effect";
import { defineRun, step, sandbox } from "@flaredispatch/core";

// Webhook mode: a multi-agent AI reviewer on every PR push — one
// tightly-scoped agent per domain, fanned out in parallel.
const AGENTS = ["security", "performance", "code-quality"] as const;

export const prReview = defineRun({
  name: "pr-review",
  version: "2.0.0",

  triggers: [
    {
      event: "pull_request",
      actions: ["opened", "synchronize"],
      idempotencyKey: ({ payload }) =>
        `pr-review:${payload.pull_request.head.sha}`,
      inputs: ({ payload }) => ({ pr: payload.pull_request.number }),
    },
  ],

  inputs: Schema.Struct({ pr: Schema.Number }),
  outputs: Schema.Struct({ verdict: Schema.String }),
  limits: { maxDurationSec: 1500, maxConcurrency: AGENTS.length },

  run: () =>
    Effect.gen(function* () {
      // Fan out one reviewer per domain, all at once.
      yield* step("review", () =>
        Effect.forEach(
          AGENTS,
          (agent) =>
            sandbox.exec({ command: ["review-agent", "run", agent] }),
          { concurrency: AGENTS.length },
        ),
      );
      // The coordinator dedups the findings into one verdict.
      const review = yield* step("coordinate", () =>
        sandbox.exec({ command: ["review-agent", "coordinate", "--json"] }),
      );
      return JSON.parse(review.stdout);
    }),
});
